📡 Flatfield Preprocessing

Blank-page flatfield scan residuals → PRNU fingerprint source extraction. Captures each scanner's unique noise pattern.

What this does: For each scanner's blank-page flatfield scans, compute residual = image − denoised. These residuals capture the scanner's unique noise pattern (PRNU fingerprint source) and are saved for fingerprint averaging in Cell 6.
Pipeline Steps
Read Image
Load TIF/PNG/JPG scan. Convert to grayscale if needed.
Resize
Resize to 256×256 using INTER_AREA for downsampling quality.
Normalize
Convert to float32, scale to [0, 1] range.
Denoise
Apply wavelet denoising (or Wiener filter) to isolate smooth base.
Residual
residual = image − denoised. This is the noise component.
Save PKL
Store all residuals per scanner in a dictionary → .pkl.
Parameters
IMG_SIZE(256, 256)
DENOISE_METHOD"wavelet" default alt: "wiener"
Input folderdata/Flatfield/{scanner}/
Outputartifacts/flatfield_residuals.pkl
Output format{ scanner_name: [np.float32 256×256, ...] }
Source Code (Cell 3)
# ============================================================
# CELL 3 — Preprocess Flatfield images → save flatfield_residuals.pkl
# Output:  artifacts/flatfield_residuals.pkl
# ============================================================

import os, pickle, cv2, numpy as np
from tqdm import tqdm
from skimage.restoration import denoise_wavelet
from scipy.signal import wiener as scipy_wiener

FLATFIELD_DIR  = "data/Flatfield"
OUTPUT_PKL     = "artifacts/flatfield_residuals.pkl"
IMG_SIZE       = (256, 256)
DENOISE_METHOD = "wavelet"   # "wiener" or "wavelet"

def preprocess_image_residual(path, img_size=IMG_SIZE, method=DENOISE_METHOD):
    img = cv2.imread(path, cv2.IMREAD_UNCHANGED)
    if img is None: return None
    if img.ndim == 3:
        img = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    img = cv2.resize(img, img_size, interpolation=cv2.INTER_AREA)
    img = img.astype(np.float32) / 255.0
    if method == "wiener":
        denoised = scipy_wiener(img, mysize=(5, 5))
    else:
        denoised = denoise_wavelet(img, channel_axis=None, rescale_sigma=True)
    return (img - denoised).astype(np.float32)

flatfield_residuals = {}
scanners = sorted(os.listdir(FLATFIELD_DIR))

for scanner in tqdm(scanners, desc="Scanners"):
    scanner_dir = os.path.join(FLATFIELD_DIR, scanner)
    if not os.path.isdir(scanner_dir): continue
    residuals = []
    for fname in sorted(os.listdir(scanner_dir)):
        if fname.startswith("._"): continue
        if not fname.lower().endswith((".tif",".tiff",".png",".jpg")): continue
        res = preprocess_image_residual(os.path.join(scanner_dir, fname))
        if res is not None: residuals.append(res)
    if residuals:
        flatfield_residuals[scanner] = residuals

os.makedirs("artifacts", exist_ok=True)
with open(OUTPUT_PKL, "wb") as f:
    pickle.dump(flatfield_residuals, f)

total = sum(len(v) for v in flatfield_residuals.values())
print(f"✅ {len(flatfield_residuals)} scanners, {total} residuals saved")
