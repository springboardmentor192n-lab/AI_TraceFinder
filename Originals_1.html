Cell 16 — PDF → TIFF Conversion
# CELL 16 — Convert original PDFs to TIFF at 300 DPI
# Requires: pip install pymupdf
import os, glob, fitz
from pathlib import Path
from PIL import Image
from tqdm import tqdm

ROOT     = "data"
CONV_OUT = "data/Originals_tif"

def pdf_to_tiffs(pdf_path, out_dir, dpi=300):
    os.makedirs(out_dir, exist_ok=True)
    doc = fitz.open(pdf_path)
    for i, page in enumerate(doc):
        pix  = page.get_pixmap(dpi=dpi)
        mode = "RGB" if pix.alpha == 0 else "RGBA"
        img  = Image.frombytes(mode, [pix.width, pix.height], pix.samples)
        if img.mode != "RGB": img = img.convert("RGB")
        stem = Path(pdf_path).stem
        img.save(f"{out_dir}/{stem}_p{i+1}.tif",
                 format="TIFF", dpi=(dpi,dpi), compression="tiff_deflate")
    doc.close()

os.makedirs(CONV_OUT, exist_ok=True)
tasks = []
for sub in ["official", "wikipedia"]:
    for pdf in glob.glob(f"{ROOT}/Originals/{sub}/**/*.pdf", recursive=True):
        parent  = Path(pdf).parent.name
        out_dir = f"{CONV_OUT}/{sub}/{parent}"
        tasks.append((pdf, out_dir))

for pdf, outd in tqdm(tasks, desc="PDF→TIFF 300dpi"):
    try: pdf_to_tiffs(pdf, outd)
    except Exception as e: print("Failed:", pdf, "->", e)
print("✅ PDF conversion complete.")
Cell 17 — Build Tamper Manifest CSV
# CELL 17 — Build tamper_manifest_grouped.csv
# Columns: path, label(0=clean/1=tampered), domain, tamper_type, page_id
import re, os, glob, csv
from pathlib import Path

ROOT      = "data"
TAMP_ROOT = f"{ROOT}/Tampered images"
MANIFEST  = "artifacts/manifests/tamper_manifest_grouped.csv"
os.makedirs(os.path.dirname(MANIFEST), exist_ok=True)

def list_tifs(root):
    return sorted([p for p in glob.glob(f"{root}/**/*", recursive=True)
                   if p.lower().endswith((".tif",".tiff"))])

def infer_page_id(p):
    m = re.search(r"(s\d+_\d+)", os.path.basename(p))
    return m.group(1) if m else re.sub(r"_p\d+$", "", Path(p).stem)

rows = [["path","label","domain","tamper_type","page_id"]]
# Clean originals
for p in list_tifs("data/Originals_tif/official"):
    rows.append([p, 0, "orig_pdf_tif", "clean", infer_page_id(p)])
for p in list_tifs("data/Originals_tif/wikipedia"):
    rows.append([p, 0, "orig_pdf_tif", "clean", infer_page_id(p)])
for p in list_tifs(f"{TAMP_ROOT}/Original"):
    rows.append([p, 0, "tamper_dir",   "clean", infer_page_id(p)])
# Tampered images
for p in list_tifs(f"{TAMP_ROOT}/Tampered/Copy-move"):
    rows.append([p, 1, "tamper_dir",   "copy-move", infer_page_id(p)])
for p in list_tifs(f"{TAMP_ROOT}/Tampered/Retouching"):
    rows.append([p, 1, "tamper_dir",   "retouch",   infer_page_id(p)])
for p in list_tifs(f"{TAMP_ROOT}/Tampered/Splicing"):
    rows.append([p, 1, "tamper_dir",   "splice",    infer_page_id(p)])

with open(MANIFEST, "w", newline="") as fh:
    csv.writer(fh).writerows(rows)
print(f"✅ Manifest: {MANIFEST} | {len(rows)-1} rows")
Dependency: pip install pymupdf is required for Cell 16. If PyMuPDF is not available, pre-render your PDFs to TIFF externally at 300 DPI and place them in data/Originals_tif/.
