Cell 4 — Residual Preprocessing
# CELL 4 — Preprocess Official + Wikipedia → official_wiki_residuals.pkl
import os, cv2, numpy as np, pickle, pywt
from concurrent.futures import ThreadPoolExecutor, as_completed

OFFICIAL_DIR = "data/Official"
WIKI_DIR     = "data/Wikipedia"
OUT_PATH     = "artifacts/official_wiki_residuals.pkl"

def denoise_wavelet_haar(img):
    cA, (cH, cV, cD) = pywt.dwt2(img, 'haar')
    cH[:] = 0; cV[:] = 0; cD[:] = 0
    return pywt.idwt2((cA, (cH, cV, cD)), 'haar')

def process_single_image(fpath):
    img = cv2.imread(fpath, cv2.IMREAD_COLOR)
    if img is None: return None
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    gray = cv2.resize(gray, (256, 256), interpolation=cv2.INTER_AREA)
    gray = gray.astype(np.float32) / 255.0
    return (gray - denoise_wavelet_haar(gray)).astype(np.float32)

def process_dataset(base_dir, dataset_name, residuals_dict):
    residuals_dict[dataset_name] = {}
    valid_exts = ('.tif', '.tiff', '.jpg', '.jpeg', '.png')
    for scanner in sorted(os.listdir(base_dir)):
        scanner_path = os.path.join(base_dir, scanner)
        if not os.path.isdir(scanner_path): continue
        residuals_dict[dataset_name][scanner] = {}
        for dpi in sorted(os.listdir(scanner_path)):
            dpi_path = os.path.join(scanner_path, dpi)
            if not os.path.isdir(dpi_path): continue
            files = [os.path.join(dpi_path, f) for f
                     in os.listdir(dpi_path) if f.lower().endswith(valid_exts)]
            dpi_res = []
            with ThreadPoolExecutor(max_workers=8) as ex:
                for fut in as_completed({ex.submit(process_single_image, f): f
                                            for f in files}):
                    r = fut.result()
                    if r is not None: dpi_res.append(r)
            residuals_dict[dataset_name][scanner][dpi] = dpi_res

residuals_dict = {}
process_dataset(OFFICIAL_DIR, "Official",  residuals_dict)
process_dataset(WIKI_DIR,     "Wikipedia", residuals_dict)

os.makedirs("artifacts", exist_ok=True)
with open(OUT_PATH, "wb") as f: pickle.dump(residuals_dict, f)
print(f"✅ Saved to {OUT_PATH}")
Cell 5 — Dataset Inventory CSV
# CELL 5 — Build dataset_inventory.csv
import os, csv

roots = [
    ("data/Official",   "Official"),
    ("data/Wikipedia",  "Wikipedia"),
]
rows = []
for root, ds_name in roots:
    for scanner in sorted(os.listdir(root)):
        for dpi in sorted(os.listdir(os.path.join(root, scanner))):
            for fname in sorted(os.listdir(os.path.join(root, scanner, dpi))):
                if fname.lower().endswith(('.tif','.tiff','.png','.jpg')):
                    rows.append([ds_name, scanner, dpi,
                                 os.path.join(root, scanner, dpi, fname)])

out_csv = "artifacts/dataset_inventory.csv"
with open(out_csv, "w", newline="") as fh:
    w = csv.writer(fh)
    w.writerow(["dataset", "scanner", "dpi", "path"])
    w.writerows(rows)
print(f"✅ {len(rows)} images indexed")

# CELL 8 — PRNU correlation vectors → features.pkl
# Result: (4400, 11) feature matrix
import pickle, numpy as np

def corr2d(a, b):
    """Zero-mean normalized cross-correlation (ZNCC)."""
    a = a.astype(np.float32).ravel(); a -= a.mean()
    b = b.astype(np.float32).ravel(); b -= b.mean()
    denom = np.linalg.norm(a) * np.linalg.norm(b)
    return float((a @ b) / denom) if denom != 0 else 0.0

with open("artifacts/official_wiki_residuals.pkl", "rb") as f:
    residuals_dict = pickle.load(f)
with open("artifacts/scanner_fingerprints.pkl", "rb") as f:
    scanner_fps = pickle.load(f)
fp_keys = np.load("artifacts/fp_keys.npy", allow_pickle=True).tolist()

features, labels = [], []
for dataset_name in ["Official", "Wikipedia"]:
    for scanner, dpi_dict in residuals_dict[dataset_name].items():
        for dpi, res_list in dpi_dict.items():
            for res in res_list:
                features.append([corr2d(res, scanner_fps[k]) for k in fp_keys])
                labels.append(scanner)

with open("artifacts/features.pkl", "wb") as f:
    pickle.dump({"features": features, "labels": labels}, f)
print(f"✅ Feature matrix: {len(features)} × {len(features[0])}")
